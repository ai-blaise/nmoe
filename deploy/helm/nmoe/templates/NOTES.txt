=== SGLang Inference Server ===

1. Get the SGLang service URL:
{{- if .Values.sglang.ingress.enabled }}
   http{{ if .Values.sglang.ingress.tls }}s{{ end }}://{{ (index .Values.sglang.ingress.hosts 0).host }}
{{- else if eq .Values.sglang.service.type "LoadBalancer" }}
   export SERVICE_IP=$(kubectl get svc {{ include "nmoe.sglang.fullname" . }} -n {{ .Values.global.namespace | default .Release.Namespace }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
   echo "SGLang URL: http://$SERVICE_IP:{{ .Values.sglang.service.port }}"
{{- else if eq .Values.sglang.service.type "NodePort" }}
   export NODE_PORT=$(kubectl get svc {{ include "nmoe.sglang.fullname" . }} -n {{ .Values.global.namespace | default .Release.Namespace }} -o jsonpath='{.spec.ports[0].nodePort}')
   export NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[?(@.type=="ExternalIP")].address}')
   echo "SGLang URL: http://$NODE_IP:$NODE_PORT"
{{- else }}
   kubectl port-forward svc/{{ include "nmoe.sglang.fullname" . }} {{ .Values.sglang.service.port }}:{{ .Values.sglang.service.port }} -n {{ .Values.global.namespace | default .Release.Namespace }}
   echo "SGLang URL: http://localhost:{{ .Values.sglang.service.port }}"
{{- end }}

2. Test the server:
   curl -X POST "http://localhost:{{ .Values.sglang.service.port }}/generate" \
     -H "Content-Type: application/json" \
     -d '{"text": "Hello, world!", "sampling_params": {"max_new_tokens": 100}}'

{{- end }}
